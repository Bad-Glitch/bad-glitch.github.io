// root-detection-bypass.js
Java.perform(function () {
    var fileClass = Java.use('java.io.File');
    fileClass.exists.implementation = function () {
        var path = this.getAbsolutePath();
        var suspicious = ['su', 'Superuser.apk', 'magisk', 'busybox'];
        for (var i = 0; i < suspicious.length; i++) {
            if (path.indexOf(suspicious[i]) !== -1) {
                console.log('[+] Hid file check: ' + path);
                return false;
            }
        }
        return this.exists();
    };

    var buildClass = Java.use('android.os.Build');
    // لو التطبيق بيفحص ro.build.tags == "test-keys" عبر System.getProperty
    var SystemClass = Java.use('java.lang.System');
    SystemClass.getProperty.overload('java.lang.String').implementation = function (key) {
        if (key === 'ro.build.tags') return 'release-keys';
        return this.getProperty(key);
    };

    // تعطيل Runtime.exec("su") المباشر
    var Runtime = Java.use('java.lang.Runtime');
    Runtime.exec.overload('java.lang.String').implementation = function (cmd) {
        if (cmd.indexOf('su') !== -1) {
            console.log('[+] Blocked Runtime.exec: ' + cmd);
            throw new (Java.use('java.io.IOException'))('not found');
        }
        return this.exec(cmd);
    };
});

09سكريبتات فكّ التعتيم بـ Python

XOR بمفتاح معروف جزئيًا (Known-Plaintext)

import base64

ciphertext = base64.b64decode("AQU6AuLW")
known_plain = b"https://"
key = bytes(c ^ p for c, p in zip(ciphertext, known_plain))
print("key candidate:", key)

def xor_decrypt(data: bytes, key: bytes) -> bytes:
    return bytes(b ^ key[i % len(key)] for i, b in enumerate(data))

print(xor_decrypt(ciphertext, key))

اكتشاف طول مفتاح XOR متكرر (Index of Coincidence)

from itertools import combinations

def hamming_distance(a: bytes, b: bytes) -> int:
    return sum(bin(x ^ y).count("1") for x, y in zip(a, b))

def guess_keysize(data: bytes, max_len=40) -> int:
    scores = {}
    for size in range(2, max_len):
        chunks = [data[i:i+size] for i in range(0, len(data) - size*2, size)][:4]
        pairs = list(combinations(chunks, 2))
        norm_dist = sum(hamming_distance(a, b) / size for a, b in pairs) / len(pairs)
        scores[size] = norm_dist
    return min(scores, key=scores.get)  # أقل مسافة = أرجح طول مفتاح

print("likely key length:", guess_keysize(open("captured.bin","rb").read()))

Brute-force لمفتاح XOR قصير (byte واحد أو كلمة سر قصيرة)

import itertools, string

def score_english(b: bytes) -> int:
    common = b" etaoinETAOIN"
    return sum(b.count(c) for c in common)

ciphertext = bytes.fromhex("1b37373331363f78151b7f2b783431333d78397828372d363c78373e783a393b3736")

14Pipeline شامل يجمع كل الخطوات

#!/usr/bin/env bash
# full_recon.sh — استطلاع وتحليل static آلي لأي APK في ثانية الأمر
set -e
APK=$1
OUT="recon_$(basename "$APK" .apk)"
mkdir -p "$OUT"

echo "[*] Hash & basic info"
sha256sum "$APK" | tee "$OUT/hash.txt"
aapt dump badging "$APK" > "$OUT/badging.txt"

echo "[*] Decompiling with jadx"
jadx -d "$OUT/jadx" "$APK" --deobf -q

echo "[*] Decoding resources with apktool"
apktool d "$APK" -o "$OUT/apktool" -f -q

echo "[*] Scanning for secrets"
grep -rEn '(?:api[_-]?key|secret|password|token)\s*[:=]\s*["\x27][0-9a-zA-Z_\-]{16,}["\x27]' "$OUT/jadx/sources" \
  > "$OUT/secrets.txt" || true

echo "[*] Scanning for weak crypto"
grep -rEn '(AES|DES|RSA)/ECB|MessageDigest.getInstance\("(MD5|SHA-1)"\)' "$OUT/jadx/sources" \
  > "$OUT/weak_crypto.txt" || true

echo "[*] Listing exported components"
grep -E 'android:exported="true"' "$OUT/apktool/AndroidManifest.xml" > "$OUT/exported.txt" || true

echo "[*] Uploading to local MobSF instance"
HASH=$(curl -s -F "file=@$APK" http://localhost:8000/api/v1/upload -H "Authorization: $MOBSF_API_KEY" | jq -r .hash)
curl -s -X POST http://localhost:8000/api/v1/scan -d "hash=$HASH" -H "Authorization: $MOBSF_API_KEY" > /dev/null
curl -s -X POST http://localhost:8000/api/v1/report_json -d "hash=$HASH" -H "Authorization: $MOBSF_API_KEY" -o "$OUT/mobsf_report.json"

echo "[+] Done. Results in $OUT/"

شغّله بـ ./full_recon.sh target_app.apk — بيطلعلك في دقايق: الأسرار المكشوفة، التشفير الضعيف، المكونات المكشوفة، وتقرير MobSF كامل، جاهزين للمراجعة اليدوية بدل ما تبدأ من صفر.
